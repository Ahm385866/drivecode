  // 6x6 ızgarada başlangıçtan hedefe engelleri aşan en kısa yolu bulur (BFS)
  function bfsPath(sx, sy, tx, ty, obstacles){
    const blocked = new Set(obstacles.map(o=>o.gx+','+o.gy));
    const key = (x,y)=>x+','+y;
    const prev = {}; prev[key(sx,sy)] = null;
    const q = [[sx,sy]]; let head = 0;
    const dirs = [[0,-1],[1,0],[0,1],[-1,0]];
    while(head < q.length){
      const [x,y] = q[head++];
      if(x===tx && y===ty) break;
      for(const [dx,dy] of dirs){
        const nx=x+dx, ny=y+dy;
        if(nx<0||ny<0||nx>=GRID||ny>=GRID) continue;
        if(blocked.has(key(nx,ny))) continue;
        if(prev[key(nx,ny)] !== undefined) continue;
        prev[key(nx,ny)] = [x,y];
        q.push([nx,ny]);
      }
    }
    if(prev[key(tx,ty)] === undefined) return null;
    const path = [];
    let cur = [tx,ty];
    while(cur){ path.unshift({gx:cur[0], gy:cur[1]}); cur = prev[key(cur[0],cur[1])]; }
    return path;
  }

        const sx=0, sy=0, tx=4, ty=4;

        const sx=0, sy=5, tx=3, ty=0;

          // 1. sıra (alt, gy=5): 3. sütun ev
          {gx:2,gy:5,type:'ev'},
          // 2. sıra (gy=4): 1 ağaç, 3 ev, 6 ağaç

          {gx:0,gy:0,type:'ağaç'}, {gx:1,gy:0,type:'ev'}, {gx:2,gy:0,type:'ağaç'}

        const path = [

      } while (Math.abs(sx-tx)+Math.abs(sy-ty) < 3 || Math.abs(sx-tx)+Math.abs(sy-ty) > 5);
 
      const manh = Math.abs(sx-tx)+Math.abs(sy-ty);
      // İlk 5 bölümde engel yok; sonraki bölümlerde engel sayısı artar; son 4 bölüm daha zor
      const obCount = i<5 ? 0 : Math.min(isLast4 ? 11 : 9, 3 + Math.floor((i-4)/1.5));

  function resize(){
    const r = cv.parentElement.getBoundingClientRect();
    cv.width = r.width * devicePixelRatio;
    cv.height = r.height * devicePixelRatio;
    ctx.setTransform(devicePixelRatio, 0, 0, devicePixelRatio, 0, 0);
    cellW = r.width / GRID;
    cellH = r.height / GRID;
    draw();
  }
  window.addEventListener('resize', resize);
 
  function roundRect(x,y,w,h,r){
    ctx.beginPath();
    ctx.moveTo(x+r,y);
    ctx.arcTo(x+w,y,x+w,y+h,r);
    ctx.arcTo(x+w,y+h,x,y+h,r);
    ctx.arcTo(x,y+h,x,y,r);
    ctx.arcTo(x,y,x+w,y,r);
    ctx.closePath();
  }
 
  function checkVisited(){
    (levels[levelIndex].checkpoints||[]).forEach((ck,idx)=>{
      if(!visited.has(idx) && Math.abs(car.gx-ck.gx)<0.5 && Math.abs(car.gy-ck.gy)<0.5){
        visited.add(idx);
        updateProgress();
      }
    });
  }
 
  function updateProgress(){
    const ckArr = levels[levelIndex].checkpoints||[];
    const done = ckArr.filter((ck,idx)=>visited.has(idx)).length;
    const pct = ckArr.length ? (done/ckArr.length*100) : 100;
    const bar = document.getElementById('progress-bar');
    if(bar) bar.style.width = pct + '%';
  }
 
  function drawCheckpoints(){
    const off = 0;
    const cs = Math.min(cellW, cellH);
    (levels[levelIndex].checkpoints||[]).forEach((ck,idx)=>{
      const px = off + ck.gx*cellW + cellW/2;
      const py = off + ck.gy*cellH + cellH/2;
      const done = visited.has(idx);
      ctx.save();
      ctx.shadowColor = done ? 'rgba(76,175,80,0.9)' : 'rgba(255,213,79,0.9)';
      ctx.shadowBlur = 12;
      ctx.fillStyle = done ? '#4caf50' : '#ffd54f';
      ctx.beginPath(); ctx.arc(px,py,cs*0.24,0,Math.PI*2); ctx.fill();
      ctx.restore();
      ctx.fillStyle = '#1a1a2e';
      ctx.font = (cs*0.32)+'px sans-serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
      ctx.fillText(done?'✓':'★', px, py);
    });
  }
 
  function drawObstacles(){
    const off = 0;
    const cs = Math.min(cellW, cellH);
    const OB_MAP = {
      'ev':    { emoji:'🏠', bg:'#8d6e63' },
      'ağaç':  { emoji:'🌳', bg:'#2e7d32' },
      'insan': { emoji:'🧍', bg:'#1565c0' }
    };
    (levels[levelIndex].obstacles||[]).forEach(ob=>{
      const px = off + ob.gx*cellW;

      const m = OB_MAP[ob.type] || OB_MAP['ev'];
      ctx.fillStyle = m.bg;
      roundRect(px+cellW*0.1, py+cellH*0.1, cellW*0.8, cellH*0.8, cs*0.1); ctx.fill();
      ctx.fillStyle = '#fff';
      ctx.font = (cs*0.4)+'px sans-serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
      ctx.fillText(m.emoji, px+cellW/2, py+cellH/2);
    });
  }
 
  function drawCar(){
    const off = 0;
    const cs = Math.min(cellW, cellH);
    const cx = off + car.gx*cellW + cellW/2;
    const cy = off + car.gy*cellH + cellH/2;
    ctx.save();
    ctx.translate(cx,cy);
    ctx.rotate(car.ang * Math.PI/180);
    const w = cs*0.52, l = cs*0.78;
    // gölge
    ctx.save();
    ctx.shadowColor = 'rgba(0,0,0,0.45)';
    ctx.shadowBlur = 10; ctx.shadowOffsetY = 4;
    ctx.fillStyle = '#e94560';
    roundRect(-w/2, -l/2, w, l, cs*0.14); ctx.fill();
    ctx.restore();
    // tekerlekler
    ctx.fillStyle = '#1a1a1a';
    const ww = cs*0.16, wl = cs*0.22;
    const wx = [-w/2-ww*0.35, w/2-ww*0.65];
    const wy = [-l/2+wl*0.15, l/2-wl*1.15];
    wx.forEach(x=>wy.forEach(y=>{ roundRect(x,y,ww,wl,cs*0.05); ctx.fill(); }));
    // gövde parıltısı
    ctx.fillStyle = '#f25c75';
    roundRect(-w/2, -l/2, w, l*0.4, cs*0.14); ctx.fill();
    // cam
    ctx.fillStyle = '#16213e';
    roundRect(-w*0.34, -l*0.18, w*0.68, l*0.5, cs*0.1); ctx.fill();
    ctx.fillStyle = 'rgba(255,255,255,0.3)';
    roundRect(-w*0.3, -l*0.32, w*0.6, l*0.16, cs*0.06); ctx.fill();
    // farlar
    ctx.fillStyle = '#ffd54f';
    ctx.beginPath(); ctx.arc(-w*0.28, -l/2+cs*0.04, cs*0.05,0,Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(w*0.28, -l/2+cs*0.04, cs*0.05,0,Math.PI*2); ctx.fill();
    ctx.restore();
  }
 
  function draw(){
    if(!car || !target) return;
    const w = cv.width, h = cv.height;
    ctx.clearRect(0,0,w,h);
    const off = 0;
    // Oyun alanı arka planı ve ızgara çizgileri
    const cs = Math.min(cellW, cellH);
    ctx.fillStyle = '#0f3460';
    ctx.fillRect(0, 0, cellW * GRID, cellH * GRID);
    ctx.strokeStyle = 'rgba(255,255,255,0.10)';
    ctx.lineWidth = Math.max(1, Math.min(cellW, cellH) * 0.025);
    for(let gx=0; gx<=GRID; gx++){
      const x = gx * cellW;
      ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, cellH * GRID); ctx.stroke();
    }
    for(let gy=0; gy<=GRID; gy++){
      const y = gy * cellH;
      ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(cellW * GRID, y); ctx.stroke();
    }
    const tx = off + target.gx*cellW + cellW/2;
    const ty = off + target.gy*cellH + cellH/2;
    // hedef parıltısı
    const pulse = 0.5 + 0.5*Math.sin(Date.now()/350);
    ctx.save();
    ctx.shadowColor = 'rgba(76,175,80,0.9)';
    ctx.shadowBlur = 18 + pulse*10;
    // bayrak direği
    ctx.strokeStyle = '#ffffff';
    ctx.lineWidth = cs*0.07;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(tx, ty + cs*0.36);
    ctx.lineTo(tx, ty - cs*0.36);
    ctx.stroke();
    // bayrak zemini
    const fw = cs*0.46, fh = cs*0.30;
    const fx = tx, fy = ty - cs*0.36;
    ctx.fillStyle = '#4caf50';
    roundRect(fx, fy, fw, fh, cs*0.05); ctx.fill();
    // dama deseni
    ctx.fillStyle = 'rgba(255,255,255,0.85)';
    const sq = fw/4;
    for(let r=0;r<2;r++) for(let c=0;c<4;c++){
      if((r+c)%2===0){ ctx.fillRect(fx + c*sq, fy + r*(fh/2), sq, fh/2); }
    }
    ctx.restore();
    drawCheckpoints();
    drawObstacles();
    drawCar();
  }
 
  function renderPanel(){
    while(panel.firstChild) panel.removeChild(panel.firstChild);

  const sleep = ms => new Promise(r=>setTimeout(r,ms));
 
  async function moveForward(){
    const steps = 20;
    const dx = Math.sin(car.ang*Math.PI/180);
    const dy = -Math.cos(car.ang*Math.PI/180);
    const obstacles = levels[levelIndex].obstacles||[];
    for(let s=0;s<steps;s++){
      car.gx += dx/steps; car.gy += dy/steps;
      checkVisited();
      for(const ob of obstacles){
        if(Math.hypot(car.gx-ob.gx, car.gy-ob.gy) < 0.5){
          draw();
          sdk.hapticFeedback('error');
          sdk.singlePlayer.actions.gameOver({ score:0, levelAttempt:{ levelIndex: levelIndex+1, stars:0 } });
          running = false;
          return false;
        }
      }
      draw(); await sleep(25);
    }
    car.gx = Math.round(car.gx); car.gy = Math.round(car.gy);
    checkVisited();
    return true;
  }
 
  async function run(){
    if(running) return;
    if(commandList.length === 0) return;
    running = true;
    sdk.hapticFeedback('medium');
    for(const cmd of commandList){
      if(cmd === 'İleri'){
        const ok = await moveForward();
        if(ok === false) return;
      } else if(cmd === 'Sağ'){
        car.ang = (car.ang + 90) % 360;
        checkVisited(); draw(); await sleep(500);
      } else if(cmd === 'Sol'){
        car.ang = (car.ang + 270) % 360;
        checkVisited(); draw(); await sleep(500);
      }
      await sleep(500);
    }
    const dist = Math.hypot(car.gx-target.gx, car.gy-target.gy);
    const ckArr = levels[levelIndex].checkpoints||[];
    const ckDone = ckArr.filter((ck,idx)=>visited.has(idx)).length;
    if(dist < 0.5){
      let stars = 1;
      if(ckArr.length === 0) stars = 3;
      else if(ckDone === ckArr.length) stars = 3;
      else if(ckDone >= Math.ceil(ckArr.length/2)) stars = 2;
      const optimal = computeSolution().length;
      const used = commandList.length;
      let score = stars * 100;
      score += Math.max(0, (optimal * 2 - used)) * 10;
      score += (levelIndex + 1) * 25;
      document.getElementById('score').textContent = score;
      sdk.hapticFeedback('success');
      sdk.singlePlayer.actions.gameOver({ score:0, levelAttempt:{ levelIndex: levelIndex+1, stars } });
    } else {
      sdk.hapticFeedback('error');
      sdk.singlePlayer.actions.gameOver({ score:0, levelAttempt:{ levelIndex: levelIndex+1, stars:0 } });
    }
    running = false;
  }
  document.getElementById('btn-run').onclick = run;
 
  function computeSolution(){
    const L = levels[levelIndex];
    const cells = L.path;
    const cmds = [];
    let ang = L.ang;
    const dirFor = (dx,dy) => {
      if(dx===0 && dy===-1) return 0;
      if(dx===1 && dy===0) return 90;
      if(dx===0 && dy===1) return 180;
      if(dx===-1 && dy===0) return 270;
      return ang;
    };
    for(let i=1;i<cells.length;i++){
      const dx = cells[i].gx - cells[i-1].gx;
      const dy = cells[i].gy - cells[i-1].gy;
      const want = dirFor(dx,dy);
      const diff = (want - ang + 360) % 360;
      if(diff===90) cmds.push('Sağ');
      else if(diff===270) cmds.push('Sol');
      else if(diff===180){ cmds.push('Sağ'); cmds.push('Sağ'); }
      cmds.push('İleri');
      ang = want;
    }
    return cmds;
  }

    controls.style.paddingLeft = ins.left + 'px';
    controls.style.paddingRight = ins.right + 'px';
    panel.style.paddingLeft = (8 + ins.left) + 'px';
    panel.style.paddingRight = (8 + ins.right) + 'px';
  }
